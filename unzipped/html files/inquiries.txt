var pulseTimers = {};
function isDark() { return !document.getElementById('phone').classList.contains('light'); }

function startPulse(arrow) {
  var id = arrow.dataset.pulseId || (arrow.dataset.pulseId = 'p' + Math.random());
  if (pulseTimers[id]) return;
  var colors = isDark() ? ['#c3563b', '#d4ec5a'] : ['#976a63', '#1a82c8'];
  var i = 0;
  arrow.style.color = colors[0]; arrow.style.opacity = '1';
  pulseTimers[id] = setInterval(function () { i = (i + 1) % 2; arrow.style.color = colors[i]; }, 600);
}

function stopPulse(arrow) {
  var id = arrow.dataset.pulseId;
  if (id && pulseTimers[id]) { clearInterval(pulseTimers[id]); delete pulseTimers[id]; }
  arrow.style.transition = 'none';
  arrow.style.color = 'var(--ink-mute)';
  arrow.style.opacity = '0.35';
}

(function () {
  var panel = document.getElementById('inq-panel');
  var header = document.getElementById('inq-header');
  var scrim = document.getElementById('panel-scrim');
  var startY = null, startX = null, curY = 0, dragging = false;

  function applyDrag(dy) {
    panel.style.transform = 'translateY(' + dy + 'px)';
    scrim.style.opacity = String(1 - Math.min(Math.max(dy / (panel.offsetHeight || 600), 0), 1));
  }
  function onStart(e) { var t = e.touches ? e.touches[0] : e; startY = t.clientY; startX = t.clientX; dragging = false; curY = 0; panel.style.transition = 'none'; scrim.style.transition = 'none'; }
  function onMove(e) {
    if (startY === null) return;
    var t = e.touches ? e.touches[0] : e;
    var dy = t.clientY - startY, dx = Math.abs(t.clientX - startX);
    if (!dragging) { if (Math.abs(dy) < 5 && dx < 5) return; if (dx > Math.abs(dy)) { startY = null; return; } dragging = true; }
    if (dy <= 0) return; curY = dy; applyDrag(curY); if (e.cancelable) e.preventDefault();
  }
  function onEnd() {
    if (!dragging) { startY = null; return; } startY = null; dragging = false;
    var h = panel.offsetHeight || 600;
    if (curY > h * 0.35) {
      var ms = Math.max((h - curY) / h * 280, 80);
      panel.style.transition = 'transform ' + ms + 'ms cubic-bezier(0.4,0,1,1)';
      scrim.style.transition = 'opacity ' + ms + 'ms linear';
      applyDrag(h);
      setTimeout(function () {
        panel.style.transform = ''; panel.style.transition = ''; scrim.style.opacity = ''; scrim.style.transition = '';
        panel.classList.remove('open', 'closing'); scrim.classList.remove('open'); scrim.style.pointerEvents = 'none';
        document.getElementById('nav-inq').classList.remove('active');
        document.getElementById('status-label').textContent = 'home';
      }, ms);
    } else {
      var ms = Math.max(curY / h * 280, 80);
      panel.style.transition = 'transform ' + ms + 'ms cubic-bezier(0.16,1,0.3,1)';
      scrim.style.transition = 'opacity ' + ms + 'ms ease';
      applyDrag(0);
      setTimeout(function () { panel.style.transition = ''; scrim.style.transition = ''; }, ms);
    }
  }
  header.addEventListener('touchstart', onStart, { passive: true });
  header.addEventListener('touchmove', onMove, { passive: false });
  header.addEventListener('touchend', onEnd);
  header.addEventListener('mousedown', onStart);
  window.addEventListener('mousemove', onMove);
  window.addEventListener('mouseup', onEnd);
})();
