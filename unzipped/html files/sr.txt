/* panel swipe to close */
(function () {
  var panel = document.getElementById('sr-panel'), header = document.getElementById('sr-header');
  var scrim = document.getElementById('panel-scrim');
  var startY = null, startX = null, curY = 0, dragging = false;
  function applyDrag(dy) { panel.style.transform = 'translateY(' + dy + 'px)'; scrim.style.opacity = String(1 - Math.min(Math.max(dy / (panel.offsetHeight || 600), 0), 1)); }
  function onStart(e) { var t = e.touches ? e.touches[0] : e; startY = t.clientY; startX = t.clientX; dragging = false; curY = 0; panel.style.transition = 'none'; scrim.style.transition = 'none'; }
  function onMove(e) {
    if (startY === null) return;
    var t = e.touches ? e.touches[0] : e, dy = t.clientY - startY, dx = Math.abs(t.clientX - startX);
    if (!dragging) { if (Math.abs(dy) < 5 && dx < 5) return; if (dx > Math.abs(dy)) { startY = null; return; } dragging = true; }
    if (dy <= 0) return; curY = dy; applyDrag(curY); if (e.cancelable) e.preventDefault();
  }
  function onEnd() {
    if (!dragging) { startY = null; return; } startY = null; dragging = false;
    var h = panel.offsetHeight || 600;
    if (curY > h * 0.35) {
      var ms = Math.max((h - curY) / h * 280, 80);
      panel.style.transition = 'transform ' + ms + 'ms cubic-bezier(0.4,0,1,1)';
      scrim.style.transition = 'opacity ' + ms + 'ms linear';
      applyDrag(h);
      setTimeout(function () { panel.style.transform = ''; panel.style.transition = ''; scrim.style.opacity = ''; scrim.style.transition = ''; panel.classList.remove('open', 'closing'); scrim.classList.remove('open'); document.getElementById('nav-menu').classList.remove('active'); document.getElementById('status-label').textContent = 'home'; }, ms);
    } else {
      var ms = Math.max(curY / h * 280, 80);
      panel.style.transition = 'transform ' + ms + 'ms cubic-bezier(0.16,1,0.3,1)'; scrim.style.transition = 'opacity ' + ms + 'ms ease'; applyDrag(0);
      setTimeout(function () { panel.style.transition = ''; scrim.style.transition = ''; }, ms);
    }
  }
  header.addEventListener('touchstart', onStart, { passive: true }); header.addEventListener('touchmove', onMove, { passive: false }); header.addEventListener('touchend', onEnd);
  header.addEventListener('mousedown', onStart); window.addEventListener('mousemove', onMove); window.addEventListener('mouseup', onEnd);
})();

document.getElementById('sr-close').addEventListener('click', function (e) { e.stopPropagation(); closePanel(); });
document.getElementById('panel-scrim').addEventListener('click', closePanel);

  var W = 0, startX = null, startY = null, dragging = false, curX = 0;
  function getDrw() { return activeDrawer; }
  function applyDrag(dx) {
    var drw = getDrw(); if (!drw) return;
    var p = Math.min(Math.max(dx / (W || 332), 0), 1);
    drw.style.transform = 'translateX(' + dx + 'px)';
    scrim.style.opacity = String(1 - p);
    scrim.style.backdropFilter = 'blur(' + ((1 - p) * 4) + 'px)'; scrim.style.webkitBackdropFilter = 'blur(' + ((1 - p) * 4) + 'px)';
  }
  function onStart(e) { if (!activeDrawer) return; var t = e.touches ? e.touches[0] : e; startX = t.clientX; startY = t.clientY; dragging = false; curX = 0; W = activeDrawer.offsetWidth || 332; activeDrawer.style.transition = 'none'; scrim.style.transition = 'none'; }
  function onMove(e) {
    if (startX === null) return; var t = e.touches ? e.touches[0] : e;
    var dx = t.clientX - startX, dy = Math.abs(t.clientY - startY);
    if (!dragging) { if (Math.abs(dx) < 5 && dy < 5) return; if (dy > Math.abs(dx)) { startX = null; return; } dragging = true; }
    if (dx <= 0) return; curX = dx; applyDrag(curX); if (e.cancelable) e.preventDefault();
  }
  function onEnd() {
    if (!dragging) { startX = null; return; } startX = null; dragging = false;
    if (curX > W * 0.35) {
      var ms = Math.max((W - curX) / W * 260, 80);
      if (activeDrawer) activeDrawer.style.transition = 'transform ' + ms + 'ms cubic-bezier(0.4,0,1,1)';
      scrim.style.transition = 'opacity ' + ms + 'ms linear'; applyDrag(W); setTimeout(closeDrawer, ms);
    } else {
      var ms = Math.max(curX / W * 260, 80);
      if (activeDrawer) activeDrawer.style.transition = 'transform ' + ms + 'ms cubic-bezier(0.16,1,0.3,1)';
      scrim.style.transition = 'opacity ' + ms + 'ms ease'; applyDrag(0);
      setTimeout(function () { if (activeDrawer) activeDrawer.style.transition = ''; scrim.style.transition = ''; }, ms);
    }
  }
  [document.getElementById('sr-detail-drawer'), document.getElementById('sr-form-drawer'), scrim].forEach(function (el) {
    el.addEventListener('touchstart', onStart, { passive: true });
    el.addEventListener('touchmove', onMove, { passive: false });
    el.addEventListener('touchend', onEnd);
  });
  document.getElementById('sr-detail-drawer').addEventListener('mousedown', onStart);
  window.addEventListener('mousemove', onMove); window.addEventListener('mouseup', onEnd);
})();
